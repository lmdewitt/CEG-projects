// Function 1: getsaturl
function getsaturl(yr, mon, fmt, lgnd, satdata) {
  if (satdata && satdata.length > 0) {
    let satid = null;
    
    // Find matching satellite data index
    for (let i = 0; i < satdata.length; i++) {
      if (yr <= satdata[i].yrmax && yr >= satdata[i].yrmin) {
        satid = i;
      }
    }
    
    if (satid === null) {
      console.warn("No satellite dataset found matching the given year.");
      return null;
    }
    
    const erddata = satdata[satid];
    const start_url = "https://coastwatch.pfeg.noaa.gov/erddap/griddap/";
    
    // Format month to 2 digits with leading zero
    const mon_str = String(parseInt(mon, 10)).padStart(2, '0');
    const time_str = `[(${yr}-${mon_str}-01T12:00:00Z)]`;
    
    const alt_str = erddata.hasalt ? "[(0.0)]" : "";
    const loc_str = "[(35):(20)][(-160):(-130)]&.draw=surface&.vars=longitude%7Clatitude%7C";
    const end_str = "&.colorBar=%7C%7CLinear%7C0%7C0.2%7C&.bgColor=0xffccccff&.trim=0";
    
    let url = `${start_url}${erddata.dset}.${fmt}?${erddata.varname}${time_str}${alt_str}${loc_str}${erddata.varname}${end_str}`;
    
    if (lgnd !== 1) {
      url += "&.legend=Off";
    }
    return url;
  }
}

// Function 2: getweeklyurl
function getweeklyurl(fmt, lgnd, satwkdata) {
  const erddata = satwkdata;
  
  if (erddata && erddata.dset) {
    const start_url = "https://coastwatch.pfeg.noaa.gov/erddap/griddap/";
    const time_str = "[(last)]";
    const alt_str = erddata.hasalt ? "[(0.0)]" : "";
    const loc_str = "[(35):(20)][(-160):(-130)]&.draw=surface&.vars=longitude%7Clatitude%7C";
    const end_str = "&.colorBar=%7C%7CLinear%7C0%7C0.2%7C&.bgColor=0xffccccff&.trim=0";
    
    let url = `${start_url}${erddata.dset}.${fmt}?${erddata.varname}${time_str}${alt_str}${loc_str}${erddata.varname}${end_str}`;
    
    if (lgnd !== 1) {
      url += "&.legend=Off";
    }
    
    return url;
  }
}
